## 3.1 HC-SR04 超声波测距

### 3.1.1 工作原理

HC-SR04 超声波传感器通过发射和接收超声波来测量距离：

```
工作流程：
1. Trig 引脚接收 10μs 高电平脉冲
2. 传感器发出 8 个 40kHz 超声波脉冲
3. 超声波遇到障碍物反射回来
4. Echo 引脚输出高电平，持续时间 = 声波往返时间
5. 距离 = 时间 × 声速(340m/s) / 2

距离公式：
distance(cm) = duration(μs) × 0.034 / 2
```

**规格参数：**
- 测量范围：2cm ~ 400cm
- 精度：±3mm
- 工作电压：5V
- 测量角度：15°

### 3.1.2 硬件连接

```
HC-SR04    Arduino UNO
  VCC  →   5V
  GND  →   GND
  Trig →   D9
  Echo →   D10
```

### 3.1.3 超声波测距代码

```cpp
/*
 * HC-SR04 超声波测距
 * 测量范围：2cm ~ 400cm，精度 ±3mm
 */

const int TRIG_PIN = 9;
const int ECHO_PIN = 10;

void setup() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  Serial.begin(9600);
  Serial.println("HC-SR04 超声波测距实验");
  Serial.println("测量范围: 2cm ~ 400cm");
}

void loop() {
  float distance = measureDistance();
  
  if (distance < 0) {
    Serial.println("超出测量范围！");
  } else {
    Serial.print("距离: ");
    Serial.print(distance, 1);
    Serial.println(" cm");
  }
  
  delay(200);
}

// 测量距离（返回厘米，-1 表示超出范围）
float measureDistance() {
  // 发送触发脉冲
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  
  // 读取回波时间（超时 30ms）
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  
  if (duration == 0) return -1;  // 超时，超出范围
  
  // 计算距离：声速 340m/s = 0.034cm/μs，除以2（往返）
  float distance = duration * 0.034 / 2.0;
  
  if (distance < 2 || distance > 400) return -1;
  
  return distance;
}
```

### 3.1.4 超声波距离报警

```cpp
/*
 * 超声波距离报警
 * 距离小于阈值时蜂鸣器报警，LED 闪烁
 */

const int TRIG_PIN   = 9;
const int ECHO_PIN   = 10;
const int BUZZER_PIN = 8;
const int LED_PIN    = 13;
const int THRESHOLD  = 20;  // 报警距离（cm）

void setup() {
  pinMode(TRIG_PIN,   OUTPUT);
  pinMode(ECHO_PIN,   INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN,    OUTPUT);
  Serial.begin(9600);
}

void loop() {
  float dist = measureDistance();
  
  if (dist > 0 && dist < THRESHOLD) {
    // 距离越近，报警越快
    int interval = map(dist, 2, THRESHOLD, 50, 500);
    tone(BUZZER_PIN, 1000, interval / 2);
    digitalWrite(LED_PIN, HIGH);
    delay(interval / 2);
    digitalWrite(LED_PIN, LOW);
    delay(interval / 2);
    
    Serial.print("⚠️ 警告！距离: ");
    Serial.print(dist, 1);
    Serial.println(" cm");
  } else {
    noTone(BUZZER_PIN);
    digitalWrite(LED_PIN, LOW);
    if (dist > 0) {
      Serial.print("距离: ");
      Serial.print(dist, 1);
      Serial.println(" cm  安全");
    }
    delay(200);
  }
}

float measureDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return -1;
  float d = duration * 0.034 / 2.0;
  return (d < 2 || d > 400) ? -1 : d;
}
```

## 3.2 DHT11 温湿度传感器

### 3.2.1 传感器介绍

DHT11 是一款常用的数字温湿度传感器：

| 参数 | DHT11 | DHT22（升级版） |
|------|-------|----------------|
| 温度范围 | 0~50°C | -40~80°C |
| 温度精度 | ±2°C | ±0.5°C |
| 湿度范围 | 20~90%RH | 0~100%RH |
| 湿度精度 | ±5%RH | ±2%RH |
| 采样率 | 1次/秒 | 0.5次/秒 |
| 工作电压 | 3.3~5V | 3.3~5V |

**需要安装库：** `DHT sensor library`（by Adafruit）

### 3.2.2 硬件连接

```
DHT11      Arduino UNO
  VCC  →   5V（或3.3V）
  DATA →   D2（需要 4.7kΩ 上拉电阻到 VCC）
  GND  →   GND

注：大多数模块已内置上拉电阻，可直接连接
```

### 3.2.3 温湿度读取代码

```cpp
/*
 * DHT11 温湿度传感器
 * 读取温度、湿度和热指数
 * 需要安装：DHT sensor library + Adafruit Unified Sensor
 */

#include <DHT.h>

#define DHT_PIN  2
#define DHT_TYPE DHT11  // 使用 DHT22 时改为 DHT22

DHT dht(DHT_PIN, DHT_TYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();
  
  Serial.println("DHT11 温湿度传感器实验");
  Serial.println("时间(s) | 温度(°C) | 湿度(%RH) | 热指数(°C)");
  Serial.println("--------|----------|-----------|----------");
}

void loop() {
  delay(2000);  // DHT11 采样间隔至少 2 秒
  
  float humidity    = dht.readHumidity();
  float temperature = dht.readTemperature();
  
  if (isnan(humidity) || isnan(temperature)) {
    Serial.println("读取失败！请检查传感器连接");
    return;
  }
  
  // 计算热指数（体感温度）
  float heatIndex = dht.computeHeatIndex(temperature, humidity, false);
  
  Serial.print(millis() / 1000);
  Serial.print("s\t| ");
  Serial.print(temperature, 1);
  Serial.print("°C\t| ");
  Serial.print(humidity, 1);
  Serial.print("%\t| ");
  Serial.print(heatIndex, 1);
  Serial.println("°C");
}
```

## 3.3 光敏电阻

### 3.3.1 工作原理

光敏电阻（LDR）的阻值随光照强度变化：
- **光照强**：阻值小（约 1kΩ）→ 分压高 → ADC 值大
- **光照弱**：阻值大（约 1MΩ）→ 分压低 → ADC 值小

**分压电路：**
```
5V ──[LDR]──┬──[10kΩ]── GND
             │
            A0（读取分压值）
```

### 3.3.2 硬件连接

```
光敏电阻一端 → 5V
光敏电阻另一端 → Arduino A0，同时通过 10kΩ 电阻接 GND
```

### 3.3.3 光敏电阻代码

```cpp
/*
 * 光敏电阻
 * 读取光照强度，自动判断明暗
 */

const int LDR_PIN = A0;
const int THRESHOLD = 500;  // 明暗阈值（根据实际调整）

void setup() {
  Serial.begin(9600);
  Serial.println("光敏电阻实验");
}

void loop() {
  int ldrValue = analogRead(LDR_PIN);
  
  // 将 ADC 值映射为光照百分比（0~100%）
  int lightPercent = map(ldrValue, 0, 1023, 0, 100);
  
  Serial.print("ADC值: ");
  Serial.print(ldrValue);
  Serial.print("  光照: ");
  Serial.print(lightPercent);
  Serial.print("%  状态: ");
  Serial.println(ldrValue > THRESHOLD ? "明亮" : "黑暗");
  
  delay(500);
}
```

### 3.3.4 自动夜灯

```cpp
/*
 * 自动夜灯
 * 环境变暗时自动开灯，变亮时关灯
 */

const int LDR_PIN  = A0;
const int LED_PIN  = 9;   // PWM 引脚，可调亮度
const int DARK_THRESHOLD = 400;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(9600);
  Serial.println("自动夜灯系统启动");
}

void loop() {
  int ldrValue = analogRead(LDR_PIN);
  
  if (ldrValue < DARK_THRESHOLD) {
    // 环境暗，开灯
    // 越暗灯越亮（反比关系）
    int brightness = map(ldrValue, 0, DARK_THRESHOLD, 255, 50);
    analogWrite(LED_PIN, brightness);
    Serial.print("夜间模式 - 亮度: ");
    Serial.println(brightness);
  } else {
    // 环境亮，关灯
    analogWrite(LED_PIN, 0);
    Serial.println("白天模式 - 灯关闭");
  }
  
  delay(200);
}
```

## 3.4 PIR 人体感应传感器

### 3.4.1 工作原理

PIR（被动红外）传感器通过检测人体发出的红外辐射来感应人体运动：
- 检测到运动：输出 HIGH（持续约 3~5 秒）
- 无运动：输出 LOW
- 感应距离：约 3~7 米
- 感应角度：约 120°

> ⚠️ **注意**：PIR 传感器上电后需要约 30 秒稳定时间，期间可能误触发。

### 3.4.2 硬件连接

```
PIR 模块   Arduino UNO
  VCC  →   5V
  GND  →   GND
  OUT  →   D2
```

### 3.4.3 PIR 人体感应代码

```cpp
/*
 * PIR 人体感应
 * 检测到人体运动时触发动作
 */

const int PIR_PIN = 2;
const int LED_PIN = 13;

void setup() {
  pinMode(PIR_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  
  Serial.begin(9600);
  Serial.println("PIR 人体感应实验");
  Serial.println("等待传感器稳定（30秒）...");
  
  // 等待 PIR 传感器稳定
  for (int i = 30; i > 0; i--) {
    Serial.print(i);
    Serial.print("... ");
    delay(1000);
  }
  Serial.println("\n传感器就绪！");
}

void loop() {
  int pirState = digitalRead(PIR_PIN);
  
  if (pirState == HIGH) {
    digitalWrite(LED_PIN, HIGH);
    Serial.print("[");
    Serial.print(millis() / 1000);
    Serial.println("s] 检测到人体运动！");
    delay(1000);  // 防止重复触发
  } else {
    digitalWrite(LED_PIN, LOW);
  }
  
  delay(100);
}
```

## 3.5 红外遥控

### 3.5.1 工作原理

红外遥控系统由**发射端**（遥控器）和**接收端**（红外接收模块）组成：
- 遥控器按键发送特定编码的红外信号（38kHz 载波）
- 接收模块解码后输出数字信号
- 常用协议：NEC、Sony、RC5 等

**需要安装库：** `IRremote`（版本 3.x）

### 3.5.2 硬件连接

```
红外接收模块  Arduino UNO
    VCC   →   5V
    GND   →   GND
    OUT   →   D11
```

### 3.5.3 红外遥控接收代码

```cpp
/*
 * 红外遥控接收
 * 接收并解码红外遥控信号
 * 需要安装：IRremote 库（版本 3.x）
 */

#include <IRremote.hpp>

const int IR_PIN = 11;

void setup() {
  Serial.begin(9600);
  IrReceiver.begin(IR_PIN, ENABLE_LED_FEEDBACK);
  Serial.println("红外遥控接收实验");
  Serial.println("按下遥控器按键...");
}

void loop() {
  if (IrReceiver.decode()) {
    Serial.print("协议: ");
    Serial.print(IrReceiver.decodedIRData.protocol == NEC ? "NEC" : "其他");
    Serial.print("  地址: 0x");
    Serial.print(IrReceiver.decodedIRData.address, HEX);
    Serial.print("  命令: 0x");
    Serial.print(IrReceiver.decodedIRData.command, HEX);
    Serial.print("  原始值: 0x");
    Serial.println(IrReceiver.decodedIRData.decodedRawData, HEX);
    
    // 根据按键执行不同操作
    switch (IrReceiver.decodedIRData.command) {
      case 0x45: Serial.println("按键: 电源"); break;
      case 0x46: Serial.println("按键: 音量+"); break;
      case 0x15: Serial.println("按键: 功能"); break;
      case 0x09: Serial.println("按键: 上一曲"); break;
      case 0x07: Serial.println("按键: 下一曲"); break;
      case 0x16: Serial.println("按键: 0"); break;
      case 0x19: Serial.println("按键: 音量-"); break;
      case 0x0D: Serial.println("按键: 播放"); break;
      default:   Serial.println("按键: 未知"); break;
    }
    
    IrReceiver.resume();  // 准备接收下一个信号
  }
}
```

## 3.6 声音传感器

### 3.6.1 传感器介绍

声音传感器模块包含麦克风和信号处理电路：
- **数字输出（DO）**：声音超过阈值时输出 HIGH
- **模拟输出（AO）**：输出与声音强度对应的模拟值
- 板载电位器可调节灵敏度

### 3.6.2 硬件连接

```
声音传感器  Arduino UNO
   VCC  →   5V
   GND  →   GND
   DO   →   D2（数字输出）
   AO   →   A0（模拟输出）
```

### 3.6.3 声音传感器代码

```cpp
/*
 * 声音传感器
 * 同时读取数字和模拟输出
 */

const int SOUND_DO = 2;   // 数字输出
const int SOUND_AO = A0;  // 模拟输出
const int LED_PIN  = 13;

void setup() {
  pinMode(SOUND_DO, INPUT);
  pinMode(LED_PIN,  OUTPUT);
  Serial.begin(9600);
  Serial.println("声音传感器实验");

  int digitalVal = digitalRead(SOUND_DO);

  Serial.print("\t| ");
  Serial.print(digitalVal);
  Serial.print("\t| ");
  Serial.println(digitalVal == HIGH ? "检测到声音！" : "安静");
  
  delay(100);
}
```

## 3.7 土壤湿度传感器

### 3.7.1 传感器介绍

土壤湿度传感器通过测量土壤电阻来判断湿度：
- 土壤湿润：导电性好，电阻小，ADC 值小
- 土壤干燥：导电性差，电阻大，ADC 值大

> ⚠️ **注意**：长期通电会导致探针氧化，建议测量时才通电。

### 3.7.2 硬件连接

```
土壤湿度传感器  Arduino UNO
    VCC     →   D7（通过引脚控制供电）
    GND     →   GND
    AO      →   A0
    DO      →   D2
```

### 3.7.3 土壤湿度代码

```cpp
/*
 * 土壤湿度传感器
 * 读取土壤湿度并判断是否需要浇水
 */

const int SOIL_AO  = A0;
const int SOIL_DO  = 2;
const int VCC_PIN  = 7;   // 通过引脚控制传感器供电

// 湿度等级阈值（根据实际传感器校准）
const int DRY_THRESHOLD  = 700;  // 干燥
const int WET_THRESHOLD  = 300;  // 湿润

void setup() {
  pinMode(SOIL_DO, INPUT);
  pinMode(VCC_PIN, OUTPUT);
  Serial.begin(9600);
  Serial.println("土壤湿度传感器实验");
}

void loop() {
  // 上电测量，减少探针氧化
  digitalWrite(VCC_PIN, HIGH);
  delay(100);
  
  int soilValue = analogRead(SOIL_AO);
  int doValue   = digitalRead(SOIL_DO);
  
  digitalWrite(VCC_PIN, LOW);  // 测量完毕断电
  
  // 判断湿度等级
  String status;
  if (soilValue > DRY_THRESHOLD) {
    status = "干燥 - 需要浇水！";
  } else if (soilValue < WET_THRESHOLD) {
    status = "湿润 - 水分充足";
  } else {
    status = "适中 - 正常";
  }
  
  Serial.print("ADC: ");
  Serial.print(soilValue);
  Serial.print("  DO: ");
  Serial.print(doValue);
  Serial.print("  状态: ");
  Serial.println(status);
  
  delay(2000);  // 每 2 秒测量一次
}
```

## 3.8 雨滴传感器

### 3.8.1 传感器介绍

雨滴传感器通过检测导电板上的水滴来判断是否下雨：
- 有水滴：导电性增加，ADC 值减小
- 无水滴：ADC 值较大

### 3.8.2 硬件连接

```
雨滴传感器  Arduino UNO
   VCC  →   5V
   GND  →   GND
   AO   →   A0
   DO   →   D2
```

### 3.8.3 雨滴传感器代码

```cpp
/*
 * 雨滴传感器
 * 检测雨水并输出状态
 */

const int RAIN_AO = A0;
const int RAIN_DO = 2;
const int LED_PIN = 13;

void setup() {
  pinMode(RAIN_DO, INPUT);
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(9600);
  Serial.println("雨滴传感器实验");
}

void loop() {
  int analogVal  = analogRead(RAIN_AO);
  int digitalVal = digitalRead(RAIN_DO);
  
  // 将 ADC 值转换为雨量百分比（值越小雨量越大）
  int rainPercent = map(analogVal, 1023, 0, 0, 100);
  rainPercent = constrain(rainPercent, 0, 100);
  
  String status;
  if (rainPercent < 10)       status = "晴天";
  else if (rainPercent < 40)  status = "小雨";
  else if (rainPercent < 70)  status = "中雨";
  else                        status = "大雨";
  
  digitalWrite(LED_PIN, digitalVal == LOW ? HIGH : LOW);
  
  Serial.print("ADC: ");
  Serial.print(analogVal);
  Serial.print("  雨量: ");
  Serial.print(rainPercent);
  Serial.print("%  状态: ");
  Serial.println(status);
  
  delay(500);
}
```

```
MQ-2 模块  Arduino UNO
   VCC  →   5V
   GND  →   GND
   AO   →   A0
   DO   →   D2
```

### 3.9.3 MQ-2 气体传感器代码

```cpp
/*
 * MQ-2 气体传感器
 * 检测可燃气体和烟雾浓度
 */

const int MQ2_AO   = A0;
const int MQ2_DO   = 2;
const int BUZZER   = 8;
const int LED_RED  = 13;

// 报警阈值（根据实际环境校准）
const int ALARM_THRESHOLD = 400;

void setup() {
  pinMode(MQ2_DO,  INPUT);
  pinMode(BUZZER,  OUTPUT);
  pinMode(LED_RED, OUTPUT);
  
  Serial.begin(9600);
  Serial.println("MQ-2 气体传感器实验");
  Serial.println("预热中（20秒）...");
  delay(20000);
  Serial.println("传感器就绪！");
}

void loop() {
  int gasValue  = analogRead(MQ2_AO);
  int doValue   = digitalRead(MQ2_DO);
  
  Serial.print("气体浓度: ");
  Serial.print(gasValue);
  Serial.print("  DO: ");
  Serial.print(doValue);
  
  if (gasValue > ALARM_THRESHOLD || doValue == HIGH) {
    Serial.println("  ⚠️ 警告：检测到有害气体！");
    // 报警
    tone(BUZZER, 2000, 500);
    digitalWrite(LED_RED, HIGH);
    delay(500);
    digitalWrite(LED_RED, LOW);
    delay(500);
  } else {
    Serial.println("  ✅ 空气正常");
    noTone(BUZZER);
    digitalWrite(LED_RED, LOW);
    delay(1000);
  }
}
```

## 3.10 传感器综合应用技巧

### 3.10.1 传感器数据滤波

原始传感器数据通常含有噪声，常用滤波方法：

```cpp
// 方法1：移动平均滤波
const int SAMPLES = 10;
int readings[SAMPLES];
int readIndex = 0;
long total = 0;

int smoothRead(int pin) {
  total -= readings[readIndex];
  readings[readIndex] = analogRead(pin);
  total += readings[readIndex];
  readIndex = (readIndex + 1) % SAMPLES;
  return total / SAMPLES;
}

// 方法2：中值滤波（去除异常值）
int medianRead(int pin) {
  int vals[5];
  for (int i = 0; i < 5; i++) {
    vals[i] = analogRead(pin);
    delay(2);
  }
  // 简单排序
  for (int i = 0; i < 4; i++)
    for (int j = i+1; j < 5; j++)
      if (vals[i] > vals[j]) { int t = vals[i]; vals[i] = vals[j]; vals[j] = t; }
  return vals[2];  // 返回中间值
}
```

### 3.10.2 传感器校准

```cpp
// 传感器校准示例（以光敏电阻为例）
int sensorMin = 1023;  // 最小值（最暗）
int sensorMax = 0;     // 最大值（最亮）

void calibrate() {
  Serial.println("校准中（5秒）：请改变光照条件...");
  unsigned long startTime = millis();
  while (millis() - startTime < 5000) {
    int val = analogRead(A0);
    sensorMin = min(sensorMin, val);
    sensorMax = max(sensorMax, val);
  }
  Serial.print("校准完成 - 最小: ");
  Serial.print(sensorMin);
  Serial.print("  最大: ");
  Serial.println(sensorMax);
}

int readCalibrated() {
  int raw = analogRead(A0);
  return map(raw, sensorMin, sensorMax, 0, 100);
}
```
