## 4.1 LCD1602 液晶显示屏

### 4.1.1 LCD1602 介绍

LCD1602 是一款常用的字符型液晶显示屏：
- 显示规格：2 行 × 16 字符
- 字符大小：5×8 点阵
- 接口：并行（16 引脚）或 I2C（通过 PCF8574 转接板）
- 工作电压：5V

**推荐使用 I2C 版本**，只需 4 根线（VCC、GND、SDA、SCL），大大简化接线。

### 4.1.2 I2C 地址扫描

使用 I2C 接口前，需要先确认模块的 I2C 地址（通常为 `0x27` 或 `0x3F`）：

```cpp
/*
 * I2C 地址扫描
 * 扫描所有 I2C 设备的地址
 */

#include <Wire.h>

void setup() {
  Wire.begin();
  Serial.begin(9600);
  Serial.println("I2C 地址扫描...");
}

void loop() {
  int deviceCount = 0;
  
  for (byte address = 1; address < 127; address++) {
    Wire.beginTransmission(address);
    byte error = Wire.endTransmission();
    
    if (error == 0) {
      Serial.print("发现设备，地址: 0x");
      if (address < 16) Serial.print("0");
      Serial.println(address, HEX);
      deviceCount++;
    }
  }
  
  if (deviceCount == 0) {
    Serial.println("未发现 I2C 设备");
  } else {
    Serial.print("共发现 ");
    Serial.print(deviceCount);
    Serial.println(" 个设备");
  }
  
  delay(5000);
}
```

### 4.1.3 硬件连接（I2C 版本）

```
LCD1602 I2C 模块   Arduino UNO
      VCC      →   5V
      GND      →   GND
      SDA      →   A4（SDA）
      SCL      →   A5（SCL）
```

**需要安装库：** `LiquidCrystal I2C`（by Frank de Brabander）

### 4.1.4 LCD1602 基础代码

```cpp
/*
 * LCD1602 I2C 基础显示
 * 显示文字、数字和自定义字符
 * 需要安装：LiquidCrystal I2C 库
 */

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// 初始化 LCD（地址 0x27，16列，2行）
// 如果显示不正常，尝试地址 0x3F
LiquidCrystal_I2C lcd(0x27, 16, 2);

// 自定义字符：心形
byte heart[8] = {
  0b00000,
  0b01010,
  0b11111,
  0b11111,
  0b01110,
  0b00100,
  0b00000,
  0b00000
};

void setup() {
  lcd.init();        // 初始化 LCD
  lcd.backlight();   // 开启背光
  
  // 创建自定义字符（最多 8 个，编号 0~7）
  lcd.createChar(0, heart);
  
  // 第一行显示
  lcd.setCursor(0, 0);  // 列, 行（从0开始）
  lcd.print("Hello Arduino!");
  
  // 第二行显示
  lcd.setCursor(0, 1);
  lcd.print("I ");
  lcd.write(0);  // 显示自定义字符（心形）
  lcd.print(" Making!");
  
  delay(3000);
}

void loop() {
  // 显示计时器
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Running time:");
  
  lcd.setCursor(0, 1);
  unsigned long seconds = millis() / 1000;
  lcd.print(seconds / 3600);
  lcd.print("h ");
  lcd.print((seconds % 3600) / 60);
  lcd.print("m ");
  lcd.print(seconds % 60);
  lcd.print("s  ");
  
  delay(1000);
}
```

### 4.1.5 LCD 显示传感器数据

```cpp
/*
 * LCD 显示温湿度数据
 * 结合 DHT11 传感器
 */

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);
DHT dht(2, DHT11);

void setup() {
  lcd.init();
  lcd.backlight();
  dht.begin();
  
  lcd.setCursor(0, 0);
  lcd.print("Temp & Humidity");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");
  delay(2000);
}

void loop() {
  float temp = dht.readTemperature();
  float humi = dht.readHumidity();
  
  lcd.clear();
  
  if (isnan(temp) || isnan(humi)) {
    lcd.setCursor(0, 0);
    lcd.print("Sensor Error!");
  } else {
    lcd.setCursor(0, 0);
    lcd.print("Temp: ");
    lcd.print(temp, 1);
    lcd.print((char)223);  // 度数符号 °
    lcd.print("C");
    
    lcd.setCursor(0, 1);
    lcd.print("Humi: ");
    lcd.print(humi, 1);
    lcd.print("%");
  }
  
  delay(2000);
}
```

## 4.2 OLED 显示屏（SSD1306）

### 4.2.1 OLED 介绍

OLED（有机发光二极管）显示屏具有以下优点：
- **自发光**：无需背光，对比度高
- **低功耗**：比 LCD 省电
- **视角广**：几乎 180° 可视角
- **分辨率**：128×64 像素（0.96 英寸）

常用驱动芯片：SSD1306（I2C 或 SPI 接口）

**需要安装库：** `Adafruit SSD1306` + `Adafruit GFX Library`

### 4.2.2 硬件连接

```
OLED SSD1306   Arduino UNO
    VCC    →   3.3V（或5V，看模块标注）
    GND    →   GND
    SDA    →   A4
    SCL    →   A5
```

### 4.2.3 OLED 基础显示代码

```cpp
/*
 * OLED SSD1306 基础显示
 * 显示文字、图形和动画
 * 需要安装：Adafruit SSD1306 + Adafruit GFX Library
 */

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH  128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1   // 无复位引脚

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Serial.begin(9600);
  
  // 初始化 OLED（I2C 地址 0x3C）
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED 初始化失败！");
    while (true);
  }
  
  display.clearDisplay();
  
  // 显示欢迎信息
  display.setTextSize(2);              // 字体大小（1~8）
  display.setTextColor(SSD1306_WHITE); // 白色文字
  display.setCursor(10, 10);
  display.println("Arduino");
  display.setTextSize(1);
  display.setCursor(20, 35);
  display.println("OLED Display");
  display.setCursor(15, 50);
  display.println("SSD1306 128x64");
  
  display.display();  // 刷新显示（必须调用）
  delay(3000);
}

void loop() {
  // 演示1：绘制图形
  display.clearDisplay();
  
  // 绘制矩形
  display.drawRect(0, 0, 128, 64, SSD1306_WHITE);
  // 绘制填充矩形
  display.fillRect(10, 10, 30, 20, SSD1306_WHITE);
  // 绘制圆形
  display.drawCircle(90, 32, 20, SSD1306_WHITE);
  // 绘制填充圆
  display.fillCircle(90, 32, 10, SSD1306_WHITE);
  // 绘制直线
  display.drawLine(0, 32, 60, 32, SSD1306_WHITE);
  
  display.display();
  delay(2000);
  
  // 演示2：显示传感器数据
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  
  display.setCursor(0, 0);
  display.println("=== System Info ===");
  display.print("Uptime: ");
  display.print(millis() / 1000);
  display.println("s");
  display.print("Free RAM: ~");
  display.print(freeMemory());
  display.println(" bytes");
  
  // 绘制进度条
  int progress = (millis() / 100) % 100;
  display.setCursor(0, 40);
  display.print("Progress: ");
  display.print(progress);
  display.println("%");
  display.drawRect(0, 52, 128, 10, SSD1306_WHITE);
  display.fillRect(1, 53, progress * 126 / 100, 8, SSD1306_WHITE);
  
  display.display();
  delay(100);
}

// 估算剩余内存（仅供参考）
int freeMemory() {
  extern int __heap_start, *__brkval;
  int v;
  return (int)&v - (__brkval == 0 ? (int)&__heap_start : (int)__brkval);
}
```

## 4.3 七段数码管

### 4.3.1 数码管介绍

七段数码管由 7 个 LED 段（a~g）和 1 个小数点（dp）组成，通过控制各段的亮灭来显示数字和部分字母。

```
    _
   |_|   ← a（上）
   |_|   ← b（右上）、c（右下）、d（下）、e（左下）、f（左上）、g（中）
   
段位编码（共阴极，1=亮）：
数字 0：a,b,c,d,e,f   = 0b00111111 = 0x3F
数字 1：b,c           = 0b00000110 = 0x06
数字 2：a,b,d,e,g     = 0b01011011 = 0x5B
数字 3：a,b,c,d,g     = 0b01001111 = 0x4F
数字 4：b,c,f,g       = 0b01100110 = 0x66
数字 5：a,c,d,f,g     = 0b01101101 = 0x6D
数字 6：a,c,d,e,f,g   = 0b01111101 = 0x7D
数字 7：a,b,c         = 0b00000111 = 0x07
数字 8：全部          = 0b01111111 = 0x7F
数字 9：a,b,c,d,f,g   = 0b01101111 = 0x6F
```

### 4.3.2 硬件连接（4 位数码管）

```
数码管引脚    Arduino UNO
  段 a    →   D2
  段 b    →   D3
  段 c    →   D4
  段 d    →   D5
  段 e    →   D6
  段 f    →   D7
  段 g    →   D8
  段 dp   →   D9
  位1(最左) → D10
  位2     →   D11
  位3     →   D12
  位4(最右) → D13
  
（共阴极：位选引脚接 GND 时该位显示）
（共阳极：位选引脚接 VCC 时该位显示）
```

### 4.3.3 数码管显示代码

```cpp
/*
 * 4 位七段数码管显示
 * 显示数字 0000~9999，实现计数器
 */

// 段引脚（a~g, dp）
const int SEG_PINS[] = {2, 3, 4, 5, 6, 7, 8, 9};
// 位选引脚（4位）
const int DIGIT_PINS[] = {10, 11, 12, 13};

// 共阴极数字编码（0~9）
const byte DIGITS[] = {
  0b00111111,  // 0
  0b00000110,  // 1
  0b01011011,  // 2
  0b01001111,  // 3
  0b01100110,  // 4
  0b01101101,  // 5
  0b01111101,  // 6
  0b00000111,  // 7
  0b01111111,  // 8
  0b01101111   // 9
};

int displayNumber = 0;
unsigned long lastUpdate = 0;

void setup() {
  for (int i = 0; i < 8; i++) {
    pinMode(SEG_PINS[i], OUTPUT);
  }
  for (int i = 0; i < 4; i++) {
    pinMode(DIGIT_PINS[i], OUTPUT);
    digitalWrite(DIGIT_PINS[i], HIGH);  // 共阴极：HIGH 关闭
  }
  Serial.begin(9600);
}

void loop() {
  // 每秒计数加1
  if (millis() - lastUpdate >= 1000) {
    lastUpdate = millis();
    displayNumber = (displayNumber + 1) % 10000;
    Serial.println(displayNumber);
  }
  
  // 动态扫描显示（每位显示约 5ms）
  showNumber(displayNumber);
}

// 显示4位数字
void showNumber(int num) {
  int digits[4];
  digits[0] = num / 1000;
  digits[1] = (num / 100) % 10;
  digits[2] = (num / 10) % 10;
  digits[3] = num % 10;
  
  for (int pos = 0; pos < 4; pos++) {
    // 关闭所有位
    for (int i = 0; i < 4; i++) {
      digitalWrite(DIGIT_PINS[i], HIGH);
    }
    
    // 设置段码
    byte segCode = DIGITS[digits[pos]];
    for (int seg = 0; seg < 8; seg++) {
      digitalWrite(SEG_PINS[seg], (segCode >> seg) & 1);
    }
    
    // 开启当前位（共阴极：LOW 开启）
    digitalWrite(DIGIT_PINS[pos], LOW);
    delay(5);
  }
}
```

## 4.4 点阵 LED（MAX7219）

### 4.4.1 MAX7219 介绍

MAX7219 是一款 LED 驱动芯片，可驱动 8×8 点阵 LED：
- 使用 SPI 接口，只需 3 根数据线
- 可级联多个模块
- 内置亮度控制（16 级）
- 工作电压：5V

**需要安装库：** `LedControl`

### 4.4.2 硬件连接

```
MAX7219 模块   Arduino UNO
   VCC    →   5V
   GND    →   GND
   DIN    →   D11（MOSI）
   CS     →   D10（SS）
   CLK    →   D13（SCK）
```

### 4.4.3 点阵 LED 显示代码

```cpp
/*
 * MAX7219 8×8 点阵 LED 显示
 * 显示数字、字母和动画
 * 需要安装：LedControl 库
 */

#include <LedControl.h>

// LedControl(DIN, CLK, CS, 模块数量)
LedControl lc = LedControl(11, 13, 10, 1);

// 数字 0~9 的点阵图案（8行，每行8位）
const byte NUMBERS[10][8] = {
  {0x3C,0x66,0x6E,0x76,0x66,0x66,0x3C,0x00},  // 0
  {0x18,0x38,0x18,0x18,0x18,0x18,0x7E,0x00},  // 1
  {0x3C,0x66,0x06,0x0C,0x18,0x30,0x7E,0x00},  // 2
  {0x3C,0x66,0x06,0x1C,0x06,0x66,0x3C,0x00},  // 3
  {0x0C,0x1C,0x2C,0x4C,0x7E,0x0C,0x0C,0x00},  // 4
  {0x7E,0x60,0x7C,0x06,0x06,0x66,0x3C,0x00},  // 5
  {0x1C,0x30,0x60,0x7C,0x66,0x66,0x3C,0x00},  // 6
  {0x7E,0x06,0x0C,0x18,0x30,0x30,0x30,0x00},  // 7
  {0x3C,0x66,0x66,0x3C,0x66,0x66,0x3C,0x00},  // 8
  {0x3C,0x66,0x66,0x3E,0x06,0x0C,0x38,0x00}   // 9
};

// 笑脸图案
const byte SMILE[8] = {
  0b00111100,
  0b01000010,
  0b10100101,
  0b10000001,
  0b10100101,
  0b10011001,
  0b01000010,
  0b00111100
};

void setup() {
  lc.shutdown(0, false);    // 唤醒（退出省电模式）
  lc.setIntensity(0, 8);   // 亮度（0~15）
  lc.clearDisplay(0);       // 清屏
  Serial.begin(9600);
}

void loop() {
  // 显示笑脸
  Serial.println("显示笑脸");
  showPattern(SMILE);
  delay(2000);
  
  // 依次显示数字 0~9
  Serial.println("显示数字");
  for (int i = 0; i <= 9; i++) {
    showPattern(NUMBERS[i]);
    delay(500);
  }
  
  // 滚动动画
  Serial.println("滚动动画");
  scrollAnimation();
}

// 显示 8×8 图案
void showPattern(const byte pattern[]) {
  for (int row = 0; row < 8; row++) {
    lc.setRow(0, row, pattern[row]);
  }
}

// 简单滚动动画
void scrollAnimation() {
  for (int shift = 0; shift < 8; shift++) {
    lc.clearDisplay(0);
    for (int row = 0; row < 8; row++) {
      lc.setRow(0, row, 1 << shift);
    }
    delay(100);
  }
}
```

## 4.5 显示模块对比与选择

| 特性 | LCD1602 | OLED SSD1306 | 七段数码管 | 点阵 LED |
|------|---------|--------------|-----------|---------|
| 分辨率 | 16×2 字符 | 128×64 像素 | 4 位数字 | 8×8 点 |
| 接口 | I2C/并行 | I2C/SPI | 直接 GPIO | SPI |
| 功耗 | 中（背光耗电） | 低 | 低 | 中 |
| 显示内容 | 文字/数字 | 文字/图形 | 数字 | 图案/动画 |
| 可视距离 | 近 | 近 | 远 | 远 |
| 价格 | 低 | 中 | 低 | 低 |
| 适用场景 | 状态显示 | 数据仪表盘 | 计数/时钟 | 装饰/动画 |

**选择建议：**
- 需要显示文字和数字 → **LCD1602**
- 需要显示图形和图表 → **OLED**
- 只需显示数字（时钟、计数器）→ **七段数码管**
- 需要动画效果 → **点阵 LED**
