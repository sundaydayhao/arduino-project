## 8.1 项目一：智能台灯

### 8.1.1 项目概述

**功能需求：**
1. 🌙 **光控**：环境暗时自动开灯，亮时关灯
2. 👤 **人感**：有人时开灯，无人延时 30 秒关灯
3. 🔘 **手动**：按键切换手动/自动模式
4. 🔆 **调光**：电位器调节亮度
5. 📊 **状态显示**：串口输出实时状态

**所需硬件：**
- Arduino UNO × 1
- 光敏电阻 + 10kΩ 电阻 × 1
- PIR 人体感应传感器 × 1
- 按键 × 1
- 电位器（10kΩ）× 1
- LED（模拟台灯）× 1 + 220Ω 电阻

### 8.1.2 硬件连接

```
光敏电阻（分压）→ A0
PIR 传感器      → D2
按键            → D3（内部上拉）
电位器          → A1
LED（台灯）     → D9（PWM）
```

### 8.1.3 系统流程图

```
上电初始化
    ↓
等待 PIR 稳定（30秒）
    ↓
进入主循环
    ↓
读取传感器（光照、PIR、按键、电位器）
    ↓
按键检测 → 切换手动/自动模式
    ↓
自动模式？
  是 → 光照暗 AND 有人 → 开灯
       无人超时 → 关灯
       光照亮 → 关灯
  否 → 按键控制开关
    ↓
PWM 输出控制亮度
    ↓
串口输出状态（每2秒）
```

### 8.1.4 完整代码

```cpp
/*
 * 综合项目：智能台灯
 * 功能：光控 + 人感 + 手动控制 + 调光
 */

const int LDR_PIN    = A0;  // 光敏电阻
const int PIR_PIN    = 2;   // PIR 人体感应
const int BTN_PIN    = 3;   // 按键
const int POT_PIN    = A1;  // 电位器
const int LAMP_PIN   = 9;   // LED 台灯（PWM）

// 参数设置
const int DARK_THRESHOLD  = 600;    // 光照阈值（低于此值认为暗）
const unsigned long AUTO_OFF_TIME = 30000;  // 无人后30秒关灯

// 状态变量
bool lampOn = false;
bool manualMode = false;
int brightness = 200;
unsigned long lastMotionTime = 0;
int lastBtnState = HIGH;
unsigned long lastDebounce = 0;

void setup() {
  pinMode(PIR_PIN, INPUT);
  pinMode(BTN_PIN, INPUT_PULLUP);
  pinMode(LAMP_PIN, OUTPUT);
  
  Serial.begin(9600);
  Serial.println("=== 智能台灯系统 ===");
  Serial.println("等待PIR传感器稳定(30s)...");
  delay(30000);
  Serial.println("系统就绪！");
}

void loop() {
  // 读取传感器
  int ldrValue  = analogRead(LDR_PIN);
  int potValue  = analogRead(POT_PIN);
  int pirState  = digitalRead(PIR_PIN);
  bool isDark   = (ldrValue > DARK_THRESHOLD);
  
  // 电位器调节亮度
  brightness = map(potValue, 0, 1023, 50, 255);
  
  // 按键切换手动/自动模式（带消抖）
  int btnReading = digitalRead(BTN_PIN);
  if (btnReading != lastBtnState) lastDebounce = millis();
  if ((millis() - lastDebounce) > 50 && btnReading == LOW && lastBtnState == HIGH) {
    manualMode = !manualMode;
    if (manualMode) lampOn = !lampOn;
    Serial.print("切换到");
    Serial.println(manualMode ? "手动模式" : "自动模式");
  }
  lastBtnState = btnReading;
  
  // 自动模式逻辑
  if (!manualMode) {
    if (pirState == HIGH) {
      lastMotionTime = millis();
      if (isDark) lampOn = true;
    }
    if (lampOn && (millis() - lastMotionTime > AUTO_OFF_TIME)) {
      lampOn = false;
      Serial.println("无人，自动关灯");
    }
    if (!isDark) lampOn = false;  // 环境亮时关灯
  }
  
  // 控制台灯
  analogWrite(LAMP_PIN, lampOn ? brightness : 0);
  
  // 串口状态输出（每2秒）
  static unsigned long lastPrint = 0;
  if (millis() - lastPrint > 2000) {
    lastPrint = millis();
    Serial.print("光照:");  Serial.print(ldrValue);
    Serial.print(isDark ? "(暗)" : "(亮)");
    Serial.print(" PIR:");  Serial.print(pirState ? "有人" : "无人");
    Serial.print(" 亮度:"); Serial.print(brightness);
    Serial.print(" 台灯:"); Serial.print(lampOn ? "开" : "关");
    Serial.print(" 模式:"); Serial.println(manualMode ? "手动" : "自动");
  }
  
  delay(100);
}
```

## 8.2 项目二：温度报警器

### 8.2.1 项目概述

**功能需求：**
1. 🌡️ **温湿度监测**：DHT11 实时读取温湿度

3. 💡 **状态指示**：绿灯正常，红灯报警
4. 📺 **LCD 显示**：实时显示温湿度数据
5. ⚙️ **阈值可调**：通过串口命令设置报警阈值

**所需硬件：**
- Arduino UNO × 1
- DHT11 温湿度传感器 × 1
- LCD1602 I2C × 1
- 蜂鸣器（无源）× 1
- 绿色 LED × 1，红色 LED × 1
- 220Ω 电阻 × 2

### 8.2.2 硬件连接

```
DHT11 DATA  → D2
LCD1602 SDA → A4
LCD1602 SCL → A5
蜂鸣器      → D8
绿色 LED    → D12（220Ω）
红色 LED    → D13（220Ω）
```

### 8.2.3 完整代码

```cpp
/*
 * 综合项目：温度报警器
 * 功能：温湿度监测 + 超温报警 + LCD显示
 */

#include <DHT.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define DHT_PIN  2
#define DHT_TYPE DHT11

DHT dht(DHT_PIN, DHT_TYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

const int BUZZER    = 8;
const int GREEN_LED = 12;
const int RED_LED   = 13;

// 报警阈值（可通过串口修改）
float tempThreshold = 30.0;  // 温度报警阈值（°C）
float humiThreshold = 80.0;  // 湿度报警阈值（%）

bool isAlarming = false;

void setup() {
  pinMode(BUZZER,    OUTPUT);
  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED,   OUTPUT);
  
  dht.begin();
  lcd.init();
  lcd.backlight();
  
  Serial.begin(9600);
  Serial.println("=== 温度报警器 ===");
  Serial.print("温度阈值: "); Serial.print(tempThreshold); Serial.println("°C");
  Serial.print("湿度阈值: "); Serial.print(humiThreshold); Serial.println("%");
  Serial.println("命令: temp:xx.x | humi:xx.x");
  
  lcd.setCursor(0, 0);
  lcd.print("Temp Alarm v1.0");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");
  delay(2000);
}

void loop() {
  // 处理串口命令
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd.startsWith("temp:")) {
      tempThreshold = cmd.substring(5).toFloat();
      Serial.print("温度阈值已设置为: "); Serial.println(tempThreshold);
    } else if (cmd.startsWith("humi:")) {
      humiThreshold = cmd.substring(5).toFloat();
      Serial.print("湿度阈值已设置为: "); Serial.println(humiThreshold);
    }
  }
  
  // 读取传感器（每2秒）
  static unsigned long lastRead = 0;
  if (millis() - lastRead >= 2000) {
    lastRead = millis();
    
    float temp = dht.readTemperature();
    float humi = dht.readHumidity();
    
    if (isnan(temp) || isnan(humi)) {
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Sensor Error!");
      Serial.println("传感器读取失败！");
      return;
    }
    
    // 判断是否报警
    bool tempAlarm = (temp >= tempThreshold);
    bool humiAlarm = (humi >= humiThreshold);
    isAlarming = tempAlarm || humiAlarm;
    
    // 更新 LCD
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("T:");
    lcd.print(temp, 1);
    lcd.print((char)223); lcd.print("C ");
    lcd.print(isAlarming ? "ALARM!" : "OK    ");
    
    lcd.setCursor(0, 1);
    lcd.print("H:");
    lcd.print(humi, 1);
    lcd.print("% Thr:");
    lcd.print((int)tempThreshold);
    lcd.print((char)223);
    
    // 串口输出
    Serial.print("温度: "); Serial.print(temp, 1); Serial.print("°C  ");
    Serial.print("湿度: "); Serial.print(humi, 1); Serial.print("%  ");
    Serial.println(isAlarming ? "⚠️ 报警！" : "✅ 正常");
    
    // 指示灯
    digitalWrite(GREEN_LED, isAlarming ? LOW : HIGH);
    digitalWrite(RED_LED,   isAlarming ? HIGH : LOW);
  }
  
  // 报警音（非阻塞）
  if (isAlarming) {
    static unsigned long lastBeep = 0;
    static bool beepState = false;
    if (millis() - lastBeep >= 500) {
      lastBeep = millis();
      beepState = !beepState;
      if (beepState) tone(BUZZER, 2000, 400);
      else noTone(BUZZER);
    }
  } else {
    noTone(BUZZER);
  }
}
```

### 8.3.3 完整代码

```cpp
/*
 * 综合项目：超声波避障小车
 * 功能：自动行驶 + 超声波避障
 */

// 超声波引脚
const int TRIG_PIN = 9;
const int ECHO_PIN = 10;

// 电机引脚
const int L_IN1 = 2, L_IN2 = 3, L_EN = 5;  // 左电机
const int R_IN1 = 4, R_IN2 = 7, R_EN = 6;  // 右电机

const int SPEED       = 180;  // 正常速度
const int TURN_SPEED  = 150;  // 转弯速度
const int SAFE_DIST   = 20;   // 安全距离（cm）
const int STOP_DIST   = 10;   // 停止距离（cm）

void setup() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(L_IN1, OUTPUT); pinMode(L_IN2, OUTPUT);
  pinMode(R_IN1, OUTPUT); pinMode(R_IN2, OUTPUT);
  
  Serial.begin(9600);
  Serial.println("=== 超声波避障小车 ===");
  delay(2000);
}

// 测量距离
float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 400;
  float d = duration * 0.034 / 2.0;
  return (d < 2 || d > 400) ? 400 : d;
}

// 电机控制
void setMotor(int in1, int in2, int en, int speed) {
  if (speed > 0) {
    digitalWrite(in1, HIGH); digitalWrite(in2, LOW);
  } else if (speed < 0) {
    digitalWrite(in1, LOW); digitalWrite(in2, HIGH);
    speed = -speed;
  } else {
    digitalWrite(in1, LOW); digitalWrite(in2, LOW);
  }
  analogWrite(en, constrain(speed, 0, 255));
}

void forward()  { setMotor(L_IN1,L_IN2,L_EN,SPEED);  setMotor(R_IN1,R_IN2,R_EN,SPEED); }
void backward() { setMotor(L_IN1,L_IN2,L_EN,-SPEED); setMotor(R_IN1,R_IN2,R_EN,-SPEED); }
void turnLeft() { setMotor(L_IN1,L_IN2,L_EN,-TURN_SPEED); setMotor(R_IN1,R_IN2,R_EN,TURN_SPEED); }
void turnRight(){ setMotor(L_IN1,L_IN2,L_EN,TURN_SPEED);  setMotor(R_IN1,R_IN2,R_EN,-TURN_SPEED); }
void stopCar()  { setMotor(L_IN1,L_IN2,L_EN,0); setMotor(R_IN1,R_IN2,R_EN,0); }

void loop() {
  float dist = getDistance();
  
  Serial.print("距离: ");
  Serial.print(dist, 1);
  Serial.print("cm  ");
  
  if (dist > SAFE_DIST) {
    // 安全，前进
    forward();
    Serial.println("前进");
  } else if (dist > STOP_DIST) {
    // 接近障碍，减速
    setMotor(L_IN1, L_IN2, L_EN, 100);
    setMotor(R_IN1, R_IN2, R_EN, 100);
    Serial.println("减速");
  } else {
    // 太近，避障
    Serial.println("避障！");
    stopCar();
    delay(300);
    
    backward();
    delay(500);
    
    // 随机选择转向方向
    if (random(2) == 0) {
      Serial.println("左转");
      turnLeft();
    } else {
      Serial.println("右转");
      turnRight();
    }
    delay(600);
    
    stopCar();
    delay(200);
  }
  
  delay(50);
}
```

**所需硬件：**
- Arduino UNO × 1
- 土壤湿度传感器 × 1

### 8.4.2 完整代码

```cpp
/*
 * 综合项目：智能浇花系统
 * 功能：土壤湿度检测 + 自动浇水 + 环境监测
 */

#include <DHT.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define DHT_PIN  2
DHT dht(DHT_PIN, DHT11);
LiquidCrystal_I2C lcd(0x27, 16, 2);

const int SOIL_PIN   = A0;  // 土壤湿度传感器
const int SOIL_VCC   = 7;   // 传感器供电控制
const int RELAY_PIN  = 8;   // 继电器（水泵）
const int BTN_PIN    = 3;   // 手动按键

// 参数
const int DRY_THRESHOLD  = 600;   // 干燥阈值
const int WET_THRESHOLD  = 300;   // 湿润阈值
const unsigned long WATER_TIME = 5000;   // 浇水时长（5秒）
const unsigned long MIN_INTERVAL = 60000; // 最短浇水间隔（1分钟）

// 状态
bool isPumping = false;
unsigned long pumpStartTime = 0;
unsigned long lastWaterTime = 0;

void setup() {
  pinMode(SOIL_VCC, OUTPUT);
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(BTN_PIN, INPUT_PULLUP);
  
  digitalWrite(RELAY_PIN, HIGH);  // 继电器初始关闭（低电平触发）
  
  dht.begin();
  lcd.init();
  lcd.backlight();
  
  Serial.begin(9600);
  Serial.println("=== 智能浇花系统 ===");
  
  lcd.setCursor(0, 0);
  lcd.print("Smart Watering");
  lcd.setCursor(0, 1);
  lcd.print("System v1.0");
  delay(2000);
}

// 读取土壤湿度
int readSoil() {
  digitalWrite(SOIL_VCC, HIGH);
  delay(100);
  int val = analogRead(SOIL_PIN);
  digitalWrite(SOIL_VCC, LOW);
  return val;
}

// 开始浇水
void startWatering(String reason) {
  if (!isPumping && (millis() - lastWaterTime > MIN_INTERVAL)) {
    isPumping = true;
    pumpStartTime = millis();
    digitalWrite(RELAY_PIN, LOW);  // 开启水泵
    Serial.print("开始浇水，原因: ");
    Serial.println(reason);
  }
}

// 停止浇水
void stopWatering() {
  isPumping = false;
  lastWaterTime = millis();
  digitalWrite(RELAY_PIN, HIGH);  // 关闭水泵
  Serial.println("浇水完成");
}

void loop() {
  // 检查浇水超时
  if (isPumping && (millis() - pumpStartTime >= WATER_TIME)) {
    stopWatering();
  }
  
  // 手动按键浇水
  if (digitalRead(BTN_PIN) == LOW) {
    delay(50);
    if (digitalRead(BTN_PIN) == LOW) {
      startWatering("手动触发");
      while (digitalRead(BTN_PIN) == LOW);
    }
  }
  
  // 每 5 秒读取传感器并更新显示
  static unsigned long lastUpdate = 0;
  if (millis() - lastUpdate >= 5000) {
    lastUpdate = millis();
    
    int soilVal = readSoil();
    float temp  = dht.readTemperature();
    float humi  = dht.readHumidity();
    
    // 判断土壤状态
    String soilStatus;
    if (soilVal > DRY_THRESHOLD) {
      soilStatus = "干燥";
      if (!isPumping) startWatering("土壤干燥");
    } else if (soilVal < WET_THRESHOLD) {
      soilStatus = "湿润";
    } else {
      soilStatus = "适中";
    }
    
    // 更新 LCD
    lcd.clear();
    lcd.setCursor(0, 0);
    if (!isnan(temp)) {
      lcd.print("T:");
      lcd.print(temp, 1);
      lcd.print((char)223);
      lcd.print("C H:");
      lcd.print((int)humi);
      lcd.print("%");
    }
    
    lcd.setCursor(0, 1);
    lcd.print("Soil:");
    lcd.print(soilVal);
    lcd.print(" ");
    lcd.print(isPumping ? "PUMP ON " : soilStatus);
    
    // 串口输出
    Serial.print("土壤: "); Serial.print(soilVal);
    Serial.print("("); Serial.print(soilStatus); Serial.print(")");
    if (!isnan(temp)) {
      Serial.print(" 温度: "); Serial.print(temp, 1); Serial.print("°C");
      Serial.print(" 湿度: "); Serial.print(humi, 1); Serial.print("%");
    }
    Serial.print(" 水泵: "); Serial.println(isPumping ? "运行中" : "停止");
  }
  
  delay(100);
}
```

## 8.5 项目五：RFID 门禁系统

### 8.5.1 项目概述

```

RC522 SCK  → D13
RC522 MOSI → D11
RC522 MISO → D12
RC522 RST  → D9
```

### 8.5.3 完整代码

```cpp
/*
 * 综合项目：RFID 门禁系统
 * 功能：RFID 识别 + 权限验证 + 门锁控制 + 动态授权
 */

#include <SPI.h>
#include <MFRC522.h>
#include <EEPROM.h>

#define SS_PIN  10
#define RST_PIN 9

MFRC522 rfid(SS_PIN, RST_PIN);

const int GREEN_LED = 7;
const int RED_LED   = 6;
const int BUZZER    = 8;
const int RELAY     = 5;

// EEPROM 存储授权卡片（每张卡 4 字节 UID + 1 字节有效标志）
#define MAX_CARDS    10
#define CARD_SIZE    5   // 4字节UID + 1字节标志
#define EEPROM_START 0

// 从 EEPROM 读取授权卡片
bool isAuthorized(byte* uid, byte uidSize) {
  for (int i = 0; i < MAX_CARDS; i++) {
    int addr = EEPROM_START + i * CARD_SIZE;
    if (EEPROM.read(addr + 4) != 0xFF) continue;  // 无效卡片
    
    bool match = true;
    for (int j = 0; j < 4 && j < uidSize; j++) {
      if (EEPROM.read(addr + j) != uid[j]) {
        match = false;
        break;
      }
    }
    if (match) return true;
  }
  return false;
}

// 添加授权卡片
bool addCard(byte* uid) {
  for (int i = 0; i < MAX_CARDS; i++) {
    int addr = EEPROM_START + i * CARD_SIZE;
    if (EEPROM.read(addr + 4) == 0xFF) continue;  // 已有卡片，跳过
    
    // 找到空位，写入
    for (int j = 0; j < 4; j++) {
      EEPROM.write(addr + j, uid[j]);
    }
    EEPROM.write(addr + 4, 0xFF);  // 标记为有效
    return true;
  }
  return false;  // 已满
}

// 删除授权卡片
bool removeCard(byte* uid) {
  for (int i = 0; i < MAX_CARDS; i++) {
    int addr = EEPROM_START + i * CARD_SIZE;
    if (EEPROM.read(addr + 4) != 0xFF) continue;
    
    bool match = true;
    for (int j = 0; j < 4; j++) {
      if (EEPROM.read(addr + j) != uid[j]) { match = false; break; }
    }
    if (match) {
      EEPROM.write(addr + 4, 0x00);  // 标记为无效
      return true;
    }
  }
  return false;
}

// 获取 UID 字符串
String getUIDString(byte* uid, byte size) {
  String s = "";
  for (byte i = 0; i < size; i++) {
    if (uid[i] < 0x10) s += "0";
    s += String(uid[i], HEX);
    if (i < size - 1) s += ":";
  }
  s.toUpperCase();
  return s;
}

void setup() {
  SPI.begin();
  rfid.PCD_Init();
  
  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED,   OUTPUT);
  pinMode(BUZZER,    OUTPUT);
  pinMode(RELAY,     OUTPUT);
  
  digitalWrite(RELAY, HIGH);  // 门锁初始关闭
  
  Serial.begin(9600);
  Serial.println("=== RFID 门禁系统 ===");
  Serial.println("命令: add(添加模式) | list(列出卡片) | clear(清空)");
  Serial.println("请刷卡...");
}

bool addMode = false;

void loop() {
  // 处理串口命令
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "add") {
      addMode = true;
      Serial.println("进入添加模式，请刷卡...");
    } else if (cmd == "list") {
      Serial.println("授权卡片列表：");
      for (int i = 0; i < MAX_CARDS; i++) {
        int addr = EEPROM_START + i * CARD_SIZE;
        if (EEPROM.read(addr + 4) == 0xFF) {
          Serial.print(i + 1); Serial.print(". ");
          for (int j = 0; j < 4; j++) {
            byte b = EEPROM.read(addr + j);
            if (b < 0x10) Serial.print("0");
            Serial.print(b, HEX);
            if (j < 3) Serial.print(":");
          }
          Serial.println();
        }
      }
    } else if (cmd == "clear") {
      for (int i = 0; i < MAX_CARDS * CARD_SIZE; i++) {
        EEPROM.write(EEPROM_START + i, 0x00);
      }
      Serial.println("已清空所有授权卡片");
    }
  }
  
  if (!rfid.PICC_IsNewCardPresent()) return;
  if (!rfid.PICC_ReadCardSerial()) return;
  
  byte* uid = rfid.uid.uidByte;
  byte  size = rfid.uid.size;
  String uidStr = getUIDString(uid, size);
  
  Serial.print("[");
  Serial.print(millis() / 1000);
  Serial.print("s] 刷卡: ");
  Serial.print(uidStr);
  
  if (addMode) {
    // 添加模式
    if (addCard(uid)) {
      Serial.println(" → 已添加授权");
      tone(BUZZER, 1000, 100); delay(150);
      tone(BUZZER, 1500, 100);
    } else {
      Serial.println(" → 授权列表已满！");
    }
    addMode = false;
  } else if (isAuthorized(uid, size)) {
    // 授权通过
    Serial.println(" → ✅ 授权通过，门已开启");
    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(RELAY, LOW);
    tone(BUZZER, 1000, 200);
    delay(3000);
    digitalWrite(RELAY, HIGH);
    digitalWrite(GREEN_LED, LOW);
  } else {
    // 未授权
    Serial.println(" → ❌ 未授权，拒绝访问");
    for (int i = 0; i < 3; i++) {
      digitalWrite(RED_LED, HIGH);
      tone(BUZZER, 2000, 150);
      delay(250);
      digitalWrite(RED_LED, LOW);
      delay(150);
    }
  }
  
  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
  delay(500);
}
```

## 8.6 项目开发方法论

### 8.6.1 项目开发流程

```
1. 需求分析
   └── 明确功能需求、硬件需求、性能要求

2. 硬件设计
   └── 选择合适的传感器和模块
   └── 绘制接线图
   └── 验证硬件连接

3. 软件设计
   └── 设计程序结构（状态机/模块化）
   └── 定义数据结构和接口
   └── 编写伪代码

4. 分模块实现
   └── 先实现各个功能模块
   └── 单独测试每个模块
   └── 逐步集成

5. 集成测试
   └── 功能测试
   └── 边界条件测试
   └── 长时间稳定性测试

6. 优化改进
   └── 代码优化
   └── 功耗优化
   └── 用户体验改进
```

### 8.6.2 常见问题与解决方案

| 问题 | 可能原因 | 解决方案 |
|------|---------|---------|
| 程序卡死 | `delay()` 阻塞 | 改用 `millis()` 非阻塞延时 |
| 内存不足 | 字符串占用大量 SRAM | 使用 `F()` 宏将字符串存入 Flash |
| 传感器读数不稳定 | 电源噪声/接线问题 | 加滤波电容，使用移动平均 |
| 蓝牙/WiFi 连接不稳定 | 波特率不匹配 | 检查并统一波特率设置 |
| 电机运行时复位 | 电源不足 | 电机独立供电，共地 |
| I2C 设备不响应 | 地址错误 | 运行 I2C 扫描程序确认地址 |
