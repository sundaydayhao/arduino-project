## 7.1 EEPROM 内部存储

### 7.1.1 EEPROM 介绍

EEPROM（电可擦可编程只读存储器）是 Arduino 内置的非易失性存储器：
- Arduino UNO：**1KB**（1024 字节）
- Arduino Mega：4KB
- 断电后数据**不丢失**
- 写入寿命：约 **100,000 次**（读取无限制）
- 每次只能读写 **1 字节**

> ⚠️ **注意**：EEPROM 写入次数有限，避免在 `loop()` 中频繁写入！

### 7.1.2 EEPROM 基础操作

```cpp
/*
 * EEPROM 读写基础
 * 演示字节、整数、浮点数和字符串的存储
 */

#include <EEPROM.h>

void setup() {
  Serial.begin(9600);
  Serial.println("EEPROM 读写实验");
  Serial.print("EEPROM 大小: ");
  Serial.print(EEPROM.length());
  Serial.println(" 字节");
  
  // ===== 写入数据 =====
  
  // 写入单字节（地址 0）
  EEPROM.write(0, 42);
  Serial.println("写入字节 42 到地址 0");
  
  // 写入整数（占 2 字节，地址 1~2）
  int myInt = 1234;
  EEPROM.put(1, myInt);
  Serial.print("写入整数 ");
  Serial.print(myInt);
  Serial.println(" 到地址 1");
  
  // 写入浮点数（占 4 字节，地址 3~6）
  float myFloat = 3.14159;
  EEPROM.put(3, myFloat);
  Serial.print("写入浮点数 ");
  Serial.print(myFloat);
  Serial.println(" 到地址 3");
  
  // 写入字符串（地址 10 开始）
  String myStr = "Hello";
  for (int i = 0; i < myStr.length(); i++) {
    EEPROM.write(10 + i, myStr[i]);
  }
  EEPROM.write(10 + myStr.length(), '\0');  // 字符串结束符
  Serial.print("写入字符串 '");
  Serial.print(myStr);
  Serial.println("' 到地址 10");
  
  Serial.println("\n===== 读取数据 =====");
  
  // 读取单字节
  byte readByte = EEPROM.read(0);
  Serial.print("地址 0 的字节: ");
  Serial.println(readByte);
  
  // 读取整数
  int readInt;
  EEPROM.get(1, readInt);
  Serial.print("地址 1 的整数: ");
  Serial.println(readInt);
  
  // 读取浮点数
  float readFloat;
  EEPROM.get(3, readFloat);
  Serial.print("地址 3 的浮点数: ");
  Serial.println(readFloat, 5);
  
  // 读取字符串
  char readStr[20];
  for (int i = 0; i < 20; i++) {
    readStr[i] = EEPROM.read(10 + i);
    if (readStr[i] == '\0') break;
  }
  Serial.print("地址 10 的字符串: ");
  Serial.println(readStr);
}

void loop() {
  // EEPROM 操作通常在 setup() 中完成
}
```

### 7.1.3 EEPROM 实用示例：保存设置

```cpp
/*
 * EEPROM 保存用户设置
 * 断电后设置不丢失
 */

#include <EEPROM.h>

// EEPROM 地址映射
#define ADDR_MAGIC    0   // 魔数（验证数据有效性）
#define ADDR_BRIGHTNESS 1 // 亮度设置
#define ADDR_THRESHOLD  2 // 阈值设置（2字节）
#define MAGIC_VALUE   0xAB // 魔数值

const int LED_PIN = 9;
const int POT_PIN = A0;

struct Settings {
  byte brightness;
  int threshold;
};

Settings settings;

// 从 EEPROM 加载设置
void loadSettings() {
  byte magic = EEPROM.read(ADDR_MAGIC);
  if (magic == MAGIC_VALUE) {
    // 数据有效，读取设置
    settings.brightness = EEPROM.read(ADDR_BRIGHTNESS);
    EEPROM.get(ADDR_THRESHOLD, settings.threshold);
    Serial.println("从 EEPROM 加载设置成功");
  } else {
    // 首次使用，使用默认值
    settings.brightness = 128;
    settings.threshold  = 500;
    Serial.println("使用默认设置");
    saveSettings();
  }
  Serial.print("亮度: "); Serial.println(settings.brightness);
  Serial.print("阈值: "); Serial.println(settings.threshold);
}

// 保存设置到 EEPROM
void saveSettings() {
  EEPROM.write(ADDR_MAGIC, MAGIC_VALUE);
  EEPROM.write(ADDR_BRIGHTNESS, settings.brightness);
  EEPROM.put(ADDR_THRESHOLD, settings.threshold);
  Serial.println("设置已保存到 EEPROM");
}

void setup() {
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(9600);
  loadSettings();
  
  // 应用设置
  analogWrite(LED_PIN, settings.brightness);
}

void loop() {
  // 读取电位器，更新亮度
  int potValue = analogRead(POT_PIN);
  byte newBrightness = map(potValue, 0, 1023, 0, 255);
  
  if (newBrightness != settings.brightness) {
    settings.brightness = newBrightness;
    analogWrite(LED_PIN, settings.brightness);
    
    // 使用 update() 代替 write()，只在值变化时写入，延长寿命
    EEPROM.update(ADDR_BRIGHTNESS, settings.brightness);
  }
  
  delay(100);
}
```

### 7.1.4 EEPROM 常用函数

```cpp
#include <EEPROM.h>

EEPROM.length();              // 返回 EEPROM 大小（字节）
EEPROM.read(address);         // 读取一个字节
EEPROM.write(address, value); // 写入一个字节（每次都写）
EEPROM.update(address, value);// 只在值不同时写入（推荐）
EEPROM.get(address, data);    // 读取任意类型数据
EEPROM.put(address, data);    // 写入任意类型数据

// 遍历所有地址
for (int i = 0; i < EEPROM.length(); i++) {
  byte val = EEPROM.read(i);
}

// 清空 EEPROM
for (int i = 0; i < EEPROM.length(); i++) {
  EEPROM.write(i, 0xFF);
}
```

## 7.2 SD 卡存储

### 7.2.1 SD 卡模块介绍

SD 卡模块让 Arduino 可以读写 SD 卡，实现大容量数据存储：
- 支持 FAT16/FAT32 文件系统
- 通信接口：SPI
- 工作电压：3.3V（模块内置电平转换）
- 适合存储传感器数据、日志文件等

**需要安装库：** Arduino 内置 `SD` 库（无需额外安装）

### 7.2.2 硬件连接

```
SD 卡模块    Arduino UNO
   CS    →   D10（片选）
   MOSI  →   D11
   MISO  →   D12
   SCK   →   D13
   VCC   →   5V
   GND   →   GND
```

### 7.2.3 SD 卡读写代码

```cpp
/*
 * SD 卡读写
 * 创建文件、写入数据、读取数据
 */

#include <SPI.h>
#include <SD.h>

const int CS_PIN = 10;

void setup() {
  Serial.begin(9600);
  Serial.println("SD 卡读写实验");
  
  // 初始化 SD 卡
  if (!SD.begin(CS_PIN)) {
    Serial.println("SD 卡初始化失败！");
    Serial.println("请检查：1.SD卡是否插入 2.接线是否正确 3.SD卡是否格式化为FAT32");
    while (true);
  }
  Serial.println("SD 卡初始化成功！");
  
  // ===== 写入文件 =====
  File dataFile = SD.open("data.txt", FILE_WRITE);
  if (dataFile) {
    dataFile.println("Arduino SD 卡测试");
    dataFile.println("时间,温度,湿度");
    dataFile.println("0,25.3,60.5");
    dataFile.println("1,25.5,61.0");
    dataFile.println("2,25.8,60.8");
    dataFile.close();
    Serial.println("数据写入成功！");
  } else {
    Serial.println("打开文件失败！");
  }
  
  // ===== 读取文件 =====
  Serial.println("\n读取文件内容：");
  dataFile = SD.open("data.txt");
  if (dataFile) {
    while (dataFile.available()) {
      Serial.write(dataFile.read());
    }
    dataFile.close();
  } else {
    Serial.println("打开文件失败！");
  }
  
  // ===== 列出文件 =====
  Serial.println("\nSD 卡文件列表：");
  File root = SD.open("/");
  printDirectory(root, 0);
  root.close();
}

void loop() {
  // 每 5 秒记录一次传感器数据
  static unsigned long lastLog = 0;
  if (millis() - lastLog >= 5000) {
    lastLog = millis();
    logData();
  }
}

// 记录传感器数据
void logData() {
  File logFile = SD.open("log.csv", FILE_WRITE);
  if (logFile) {
    logFile.print(millis() / 1000);
    logFile.print(",");
    logFile.print(analogRead(A0));
    logFile.print(",");
    logFile.println(analogRead(A1));
    logFile.close();
    Serial.println("数据已记录");
  }
}

// 递归列出目录
void printDirectory(File dir, int numTabs) {
  while (true) {
    File entry = dir.openNextFile();
    if (!entry) break;
    
    for (int i = 0; i < numTabs; i++) Serial.print("  ");
    Serial.print(entry.name());
    
    if (entry.isDirectory()) {
      Serial.println("/");
      printDirectory(entry, numTabs + 1);
    } else {
      Serial.print("  ");
      Serial.print(entry.size());
      Serial.println(" bytes");
    }
    entry.close();
  }
}
```

## 7.3 RTC 实时时钟（DS3231）

### 7.3.1 DS3231 介绍

DS3231 是一款高精度实时时钟模块：
- 精度：±2ppm（每天误差约 0.17 秒）
- 内置温度补偿晶振（TCXO）
- 通信接口：I2C
- 工作电压：3.3~5.5V
- 内置 CR2032 备用电池，断电后继续计时
- 可存储年、月、日、时、分、秒、星期

**需要安装库：** `RTClib`（by Adafruit）

### 7.3.2 硬件连接

```
DS3231 模块   Arduino UNO
   VCC    →   5V（或3.3V）
   GND    →   GND
   SDA    →   A4
   SCL    →   A5
   SQW    →   不接（可选，方波输出）
   32K    →   不接（可选，32kHz输出）
```

### 7.3.3 RTC 时钟代码

```cpp
/*
 * DS3231 RTC 实时时钟
 * 读取和设置时间，显示日期时间
 * 需要安装：RTClib 库（by Adafruit）
 */

#include <Wire.h>
#include <RTClib.h>

RTC_DS3231 rtc;

// 星期名称
const char* DAYS[] = {"日", "一", "二", "三", "四", "五", "六"};
const char* MONTHS[] = {"", "1月", "2月", "3月", "4月", "5月", "6月",
                         "7月", "8月", "9月", "10月", "11月", "12月"};

void setup() {
  Serial.begin(9600);
  Wire.begin();
  
  if (!rtc.begin()) {
    Serial.println("DS3231 初始化失败！请检查接线");
    while (true);
  }
  
  // 检查是否需要设置时间
  if (rtc.lostPower()) {
    Serial.println("RTC 电源丢失，正在设置时间...");
    // 设置为编译时间（上传时自动同步）
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }
  
  // 手动设置时间（取消注释并修改）
  // rtc.adjust(DateTime(2026, 3, 25, 10, 30, 0));
  
  Serial.println("DS3231 RTC 实时时钟实验");
  
  // 显示当前温度（DS3231 内置温度传感器）
  Serial.print("当前温度: ");
  Serial.print(rtc.getTemperature());
  Serial.println("°C");
}

void loop() {
  DateTime now = rtc.now();
  
  // 格式化输出：2026-03-25 10:30:00 周三
  Serial.print(now.year());
  Serial.print("-");
  if (now.month() < 10) Serial.print("0");
  Serial.print(now.month());
  Serial.print("-");
  if (now.day() < 10) Serial.print("0");
  Serial.print(now.day());
  Serial.print(" ");
  if (now.hour() < 10) Serial.print("0");
  Serial.print(now.hour());
  Serial.print(":");
  if (now.minute() < 10) Serial.print("0");
  Serial.print(now.minute());
  Serial.print(":");
  if (now.second() < 10) Serial.print("0");
  Serial.print(now.second());
  Serial.print(" 周");
  Serial.println(DAYS[now.dayOfTheWeek()]);
  
  delay(1000);
}
```

### 7.3.4 RTC + SD 卡数据记录器

将 RTC 和 SD 卡结合，实现带时间戳的数据记录：

```cpp
/*
 * 数据记录器
 * 结合 DS3231 RTC 和 SD 卡
 * 每分钟记录一次传感器数据（带时间戳）
 */

#include <Wire.h>
#include <RTClib.h>
#include <SPI.h>
#include <SD.h>

RTC_DS3231 rtc;
const int SD_CS = 10;
const int SENSOR_PIN = A0;

void setup() {
  Serial.begin(9600);
  Wire.begin();
  
  // 初始化 RTC
  if (!rtc.begin()) {
    Serial.println("RTC 初始化失败！");
    while (true);
  }
  if (rtc.lostPower()) {
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }
  
  // 初始化 SD 卡
  if (!SD.begin(SD_CS)) {
    Serial.println("SD 卡初始化失败！");
    while (true);
  }
  
  // 写入 CSV 表头
  if (!SD.exists("datalog.csv")) {
    File f = SD.open("datalog.csv", FILE_WRITE);
    if (f) {
      f.println("日期,时间,传感器值,电压(V)");
      f.close();
    }
  }
  
  Serial.println("数据记录器启动！");
  Serial.println("每分钟记录一次数据...");
}

void loop() {
  static int lastMinute = -1;
  DateTime now = rtc.now();
  
  // 每分钟记录一次
  if (now.minute() != lastMinute) {
    lastMinute = now.minute();
    
    int sensorVal = analogRead(SENSOR_PIN);
    float voltage = sensorVal * 5.0 / 1023.0;
    
    // 构建时间戳字符串
    char timestamp[30];
    sprintf(timestamp, "%04d-%02d-%02d,%02d:%02d:%02d",
            now.year(), now.month(), now.day(),
            now.hour(), now.minute(), now.second());
    
    // 写入 SD 卡
    File logFile = SD.open("datalog.csv", FILE_WRITE);
    if (logFile) {
      logFile.print(timestamp);
      logFile.print(",");
      logFile.print(sensorVal);
      logFile.print(",");
      logFile.println(voltage, 3);
      logFile.close();
      
      Serial.print("记录: ");
      Serial.print(timestamp);
      Serial.print(", ");
      Serial.print(sensorVal);
      Serial.print(", ");
      Serial.println(voltage, 3);
    }
  }
  
  delay(1000);
}
```

## 7.4 存储方案对比

| 存储方式 | 容量 | 速度 | 断电保持 | 写入寿命 | 适用场景 |
|---------|------|------|---------|---------|---------|
| EEPROM | 1KB | 慢 | ✅ | 10万次 | 配置参数、小数据 |
| SD 卡 | GB级 | 中 | ✅ | 高 | 数据日志、文件存储 |
| SRAM | 2KB | 快 | ❌ | 无限 | 运行时变量 |
| Flash | 32KB | 中 | ✅ | 1万次 | 程序代码 |

**选择建议：**
- 存储少量配置（<100字节）→ **EEPROM**
- 存储大量数据/文件 → **SD 卡**
- 需要精确时间戳 → **DS3231 RTC**
- 临时数据处理 → **SRAM（普通变量）**
