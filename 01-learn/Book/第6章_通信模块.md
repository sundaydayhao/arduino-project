## 6.1 蓝牙通信（HC-05）

### 6.1.1 HC-05 模块介绍

HC-05 是一款常用的蓝牙串口模块，支持主从模式：
- **从机模式（Slave）**：等待其他设备连接（最常用）
- **主机模式（Master）**：主动连接其他蓝牙设备
- 通信距离：约 10 米
- 工作电压：3.3V（逻辑电平），5V 供电
- 默认波特率：9600（AT 模式：38400）
- 默认配对码：1234 或 0000

### 6.1.2 AT 命令配置

HC-05 通过 AT 命令进行配置。进入 AT 模式：
1. 按住模块上的按键
2. 给模块上电
3. LED 慢速闪烁（约 2 秒一次）表示进入 AT 模式

```cpp
/*
 * HC-05 AT 命令配置
 * 通过 Arduino 发送 AT 命令配置蓝牙模块
 * 
 * 接线：
 *   HC-05 TX → Arduino D10（软串口 RX）
 *   HC-05 RX → Arduino D11（软串口 TX，需要分压到 3.3V）
 *   HC-05 VCC → 5V
 *   HC-05 GND → GND
 *   HC-05 KEY → 3.3V（进入 AT 模式）
 */

#include <SoftwareSerial.h>

SoftwareSerial btSerial(10, 11);  // RX, TX

void setup() {
  Serial.begin(9600);
  btSerial.begin(38400);  // AT 模式波特率
  
  Serial.println("HC-05 AT 命令配置");
  Serial.println("常用 AT 命令：");
  Serial.println("  AT          → 测试连接");
  Serial.println("  AT+NAME?    → 查询名称");
  Serial.println("  AT+NAME=xxx → 设置名称");
  Serial.println("  AT+PSWD?    → 查询配对码");
  Serial.println("  AT+PSWD=xxxx → 设置配对码");
  Serial.println("  AT+BAUD?    → 查询波特率");
  Serial.println("  AT+BAUD4    → 设置波特率9600");
  Serial.println("  AT+ROLE?    → 查询模式(0=从机,1=主机)");
  Serial.println("  AT+RESET    → 重启模块");
}

void loop() {
  // 将串口监视器的输入转发给蓝牙模块
  if (Serial.available()) {
    btSerial.write(Serial.read());
  }
  // 将蓝牙模块的响应转发给串口监视器
  if (btSerial.available()) {
    Serial.write(btSerial.read());
  }
}
```

### 6.1.3 硬件连接（通信模式）

```
HC-05 模块    Arduino UNO
   VCC    →   5V
   GND    →   GND
   TXD    →   D10（软串口 RX）
   RXD    →   D11（软串口 TX）通过分压电路降到 3.3V
   
分压电路（5V → 3.3V）：
   D11 → 1kΩ → HC-05 RXD
              → 2kΩ → GND
```

> ⚠️ **注意**：HC-05 的 RXD 引脚只能接受 3.3V 信号，需要分压！

### 6.1.4 蓝牙通信代码

```cpp
/*
 * HC-05 蓝牙通信
 * 通过蓝牙接收命令控制 Arduino
 * 手机端使用蓝牙串口助手 APP
 */

#include <SoftwareSerial.h>

SoftwareSerial btSerial(10, 11);  // RX=D10, TX=D11

const int LED_PIN = 13;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  
  Serial.begin(9600);
  btSerial.begin(9600);  // 与 HC-05 配置的波特率一致
  
  Serial.println("蓝牙通信实验");
  Serial.println("等待蓝牙连接...");
  btSerial.println("Arduino 蓝牙已就绪！");
  btSerial.println("命令: on/off/blink/status");
}

void loop() {
  // 接收蓝牙数据
  if (btSerial.available()) {
    String cmd = btSerial.readStringUntil('\n');
    cmd.trim();
    cmd.toLowerCase();
    
    Serial.print("蓝牙收到: ");
    Serial.println(cmd);
    
    if (cmd == "on") {
      digitalWrite(LED_PIN, HIGH);
      btSerial.println("LED 已开启");
    } else if (cmd == "off") {
      digitalWrite(LED_PIN, LOW);
      btSerial.println("LED 已关闭");
    } else if (cmd == "blink") {
      btSerial.println("LED 闪烁中...");
      for (int i = 0; i < 5; i++) {
        digitalWrite(LED_PIN, HIGH); delay(300);
        digitalWrite(LED_PIN, LOW);  delay(300);
      }
      btSerial.println("闪烁完成");
    } else if (cmd == "status") {
      btSerial.print("运行时间: ");
      btSerial.print(millis() / 1000);
      btSerial.println("秒");
    } else {
      btSerial.print("未知命令: ");
      btSerial.println(cmd);
    }
  }
  
  // 将串口数据转发给蓝牙（调试用）
  if (Serial.available()) {
    btSerial.write(Serial.read());
  }
}
```

## 6.2 WiFi 通信（ESP8266）

### 6.2.1 ESP8266 模块介绍

ESP8266 是一款低成本 WiFi 模块，可让 Arduino 连接互联网：
- 工作电压：**3.3V**（注意！不能接 5V）
- 通信接口：UART（AT 命令）
- 支持 WiFi 802.11 b/g/n
- 内置 TCP/IP 协议栈

> ⚠️ **重要**：ESP8266 工作电压为 3.3V，必须使用 3.3V 供电！

### 6.2.2 硬件连接

```
ESP8266 (ESP-01)   Arduino UNO
    VCC        →   3.3V（必须！不能用5V）
    GND        →   GND
    TX         →   D10（软串口 RX）
    RX         →   D11（软串口 TX，需分压到 3.3V）
    CH_PD      →   3.3V（使能引脚，必须接高电平）
    RST        →   3.3V（或悬空）
    
注意：ESP8266 需要较大电流（约 200mA），
     Arduino 的 3.3V 引脚可能不够，建议使用独立 3.3V 稳压模块！
```

### 6.2.3 ESP8266 WiFi 代码

```cpp
/*
 * ESP8266 WiFi 通信
 * 连接 WiFi 并发送 HTTP 请求
 * 需要：SoftwareSerial 库
 */

#include <SoftwareSerial.h>

SoftwareSerial espSerial(10, 11);  // RX, TX

const char* SSID     = "你的WiFi名称";
const char* PASSWORD = "你的WiFi密码";
const char* SERVER   = "api.thingspeak.com";  // 示例服务器

void setup() {
  Serial.begin(9600);
  espSerial.begin(115200);
  
  Serial.println("ESP8266 WiFi 实验");
  
  // 初始化 ESP8266
  sendCommand("AT", 2000);
  sendCommand("AT+CWMODE=1", 2000);  // 设置为 Station 模式
  
  // 连接 WiFi
  String connectCmd = "AT+CWJAP=\"";
  connectCmd += SSID;
  connectCmd += "\",\"";
  connectCmd += PASSWORD;
  connectCmd += "\"";
  
  Serial.println("连接 WiFi...");
  sendCommand(connectCmd, 10000);
  
  // 获取 IP 地址
  sendCommand("AT+CIFSR", 2000);
  
  Serial.println("WiFi 连接完成！");
}

void loop() {
  // 每 30 秒发送一次数据到 ThingSpeak
  static unsigned long lastSend = 0;
  if (millis() - lastSend >= 30000) {
    lastSend = millis();
    
    int sensorValue = analogRead(A0);
    sendData(sensorValue);
  }
}

// 发送 AT 命令
String sendCommand(String cmd, int timeout) {
  espSerial.println(cmd);
  
  String response = "";
  unsigned long start = millis();
  
  while (millis() - start < timeout) {
    while (espSerial.available()) {
      response += (char)espSerial.read();
    }
  }
  
  Serial.print("CMD: "); Serial.println(cmd);
  Serial.print("RSP: "); Serial.println(response);
  return response;
}

// 发送传感器数据到 ThingSpeak
void sendData(int value) {
  String httpRequest = "GET /update?api_key=YOUR_API_KEY&field1=";
  httpRequest += value;
  httpRequest += " HTTP/1.1\r\nHost: ";
  httpRequest += SERVER;
  httpRequest += "\r\nConnection: close\r\n\r\n";
  
  // 建立 TCP 连接
  String connectCmd = "AT+CIPSTART=\"TCP\",\"";
  connectCmd += SERVER;
  connectCmd += "\",80";
  sendCommand(connectCmd, 5000);
  
  // 发送数据长度
  String sendCmd = "AT+CIPSEND=";
  sendCmd += httpRequest.length();
  sendCommand(sendCmd, 2000);
  
  // 发送 HTTP 请求
  espSerial.print(httpRequest);
  delay(2000);
  
  // 关闭连接
  sendCommand("AT+CIPCLOSE", 2000);
  
  Serial.print("已发送数据: ");
  Serial.println(value);
}
```

### 6.2.4 ESP8266 常用 AT 命令

```
基础命令：
  AT              → 测试连接，返回 OK
  AT+RST          → 重启模块
  AT+GMR          → 查询固件版本
  AT+RESTORE      → 恢复出厂设置

WiFi 命令：
  AT+CWMODE=1     → Station 模式（连接路由器）
  AT+CWMODE=2     → AP 模式（创建热点）
  AT+CWMODE=3     → AP+Station 混合模式
  AT+CWLAP        → 扫描可用 WiFi
  AT+CWJAP="ssid","pwd"  → 连接 WiFi
  AT+CWQAP        → 断开 WiFi
  AT+CIFSR        → 获取 IP 地址

TCP/IP 命令：
  AT+CIPSTART="TCP","host",port  → 建立 TCP 连接
  AT+CIPSEND=n    → 发送 n 字节数据
  AT+CIPCLOSE     → 关闭连接
  AT+CIPMUX=1     → 开启多连接模式
```

## 6.3 RFID 读卡（RC522）

### 6.3.1 RFID 工作原理

RFID（射频识别）系统由**读卡器**（RC522）和**标签**（卡片/钥匙扣）组成：

```
工作流程：
1. 读卡器发出 13.56MHz 射频信号
2. 标签进入磁场后被激活
3. 标签发送唯一 UID（卡号）
4. 读卡器接收并解码 UID
5. 根据 UID 判断是否授权
```

**RC522 规格：**
- 工作频率：13.56MHz
- 通信接口：SPI
- 工作电压：3.3V
- 读取距离：约 5cm

**需要安装库：** `MFRC522`

### 6.3.2 硬件连接

```
RC522 模块    Arduino UNO
   SDA    →   D10（SS）
   SCK    →   D13（SCK）
   MOSI   →   D11（MOSI）
   MISO   →   D12（MISO）
   IRQ    →   不接
   GND    →   GND
   RST    →   D9
   3.3V   →   3.3V（必须！不能用5V）
```

### 6.3.3 RFID 读卡代码

```cpp
/*
 * RFID RC522 读卡
 * 读取 RFID 卡的 UID 并显示
 * 需要安装：MFRC522 库
 */

#include <SPI.h>
#include <MFRC522.h>

#define SS_PIN  10
#define RST_PIN 9

MFRC522 rfid(SS_PIN, RST_PIN);

void setup() {
  SPI.begin();
  rfid.PCD_Init();
  
  Serial.begin(9600);
  Serial.println("RFID RC522 读卡实验");
  Serial.println("请将卡片靠近读卡器...");
  
  // 显示读卡器版本
  rfid.PCD_DumpVersionToSerial();
}

void loop() {
  // 检测是否有新卡
  if (!rfid.PICC_IsNewCardPresent()) return;
  
  // 读取卡片
  if (!rfid.PICC_ReadCardSerial()) return;
  
  // 获取卡片类型
  MFRC522::PICC_Type piccType = rfid.PICC_GetType(rfid.uid.sak);
  Serial.print("卡片类型: ");
  Serial.println(rfid.PICC_GetTypeName(piccType));
  
  // 显示 UID
  Serial.print("UID: ");
  String uidStr = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) {
      Serial.print("0");
      uidStr += "0";
    }
    Serial.print(rfid.uid.uidByte[i], HEX);
    uidStr += String(rfid.uid.uidByte[i], HEX);
    if (i < rfid.uid.size - 1) {
      Serial.print(":");
      uidStr += ":";
    }
  }
  Serial.println();
  uidStr.toUpperCase();
  Serial.print("UID字符串: ");
  Serial.println(uidStr);
  
  // 停止读卡
  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
  
  delay(1000);
}
```

### 6.3.4 RFID 门禁系统

```cpp
/*
 * RFID 门禁系统
 * 授权卡片开门，未授权卡片报警
 */

#include <SPI.h>
#include <MFRC522.h>

#define SS_PIN  10
#define RST_PIN 9

MFRC522 rfid(SS_PIN, RST_PIN);

const int GREEN_LED = 7;   // 授权指示
const int RED_LED   = 6;   // 拒绝指示
const int BUZZER    = 8;   // 蜂鸣器
const int RELAY     = 5;   // 继电器（控制门锁）

// 授权卡片 UID 列表（先用读卡程序获取 UID）
const String AUTHORIZED_UIDS[] = {
  "A1:B2:C3:D4",  // 卡片1
  "E5:F6:07:08",  // 卡片2
};
const int AUTH_COUNT = 2;

void setup() {
  SPI.begin();
  rfid.PCD_Init();
  
  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED,   OUTPUT);
  pinMode(BUZZER,    OUTPUT);
  pinMode(RELAY,     OUTPUT);
  
  digitalWrite(RELAY, HIGH);  // 继电器初始关闭（门锁）
  
  Serial.begin(9600);
  Serial.println("=== RFID 门禁系统 ===");
  Serial.println("请刷卡...");
}

void loop() {
  if (!rfid.PICC_IsNewCardPresent()) return;
  if (!rfid.PICC_ReadCardSerial()) return;
  
  // 获取 UID 字符串
  String uid = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) uid += "0";
    uid += String(rfid.uid.uidByte[i], HEX);
    if (i < rfid.uid.size - 1) uid += ":";
  }
  uid.toUpperCase();
  
  Serial.print("刷卡 UID: ");
  Serial.println(uid);
  
  // 检查是否授权
  bool authorized = false;
  for (int i = 0; i < AUTH_COUNT; i++) {
    if (uid == AUTHORIZED_UIDS[i]) {
      authorized = true;
      break;
    }
  }
  
  if (authorized) {
    Serial.println("✅ 授权通过！门已开启");
    // 绿灯亮，开门 3 秒
    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(RELAY, LOW);   // 开门
    tone(BUZZER, 1000, 200);
    delay(3000);
    digitalWrite(RELAY, HIGH);  // 关门
    digitalWrite(GREEN_LED, LOW);
  } else {
    Serial.println("❌ 未授权！拒绝访问");
    // 红灯闪烁，蜂鸣器报警
    for (int i = 0; i < 3; i++) {
      digitalWrite(RED_LED, HIGH);
      tone(BUZZER, 2000, 200);
      delay(300);
      digitalWrite(RED_LED, LOW);
      delay(200);
    }
  }
  
  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
  delay(500);
}
```

## 6.4 通信协议对比

| 协议 | 距离 | 速度 | 功耗 | 适用场景 |
|------|------|------|------|---------|
| 蓝牙（HC-05） | ~10m | 中 | 低 | 手机控制、近距离通信 |
| WiFi（ESP8266） | ~50m | 高 | 中 | 物联网、远程控制 |
| RFID（RC522） | ~5cm | 低 | 极低 | 门禁、身份识别 |
| I2C | 板内 | 低 | 极低 | 传感器、显示屏 |
| SPI | 板内 | 高 | 低 | 高速外设 |
| UART | 短距 | 中 | 低 | 调试、模块通信 |
